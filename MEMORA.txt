  const firebaseConfig = JSON.parse(firebaseConfigStr);
  if (firebaseConfig && firebaseConfig.apiKey) {
    app = initializeApp(firebaseConfig);
    auth = getAuth(app);
    db = getFirestore(app);
  }
} catch (error) {
  console.warn("Firebase initialization skipped or failed, using local fallback state:", error);
}

export default function App() {
  const [user, setUser] = useState(null);
  const [activeTab, setActiveTab] = useState('beranda'); // 'beranda' | 'ensiklopedia' | 'buku_harian' | 'pengaturan'
  const [loading, setLoading] = useState(true);

  // Core Data States
  const [profile, setProfile] = useState(DEFAULT_PROFILE);
  const [settings, setSettings] = useState(DEFAULT_SETTINGS);
  const [encyclopedia, setEncyclopedia] = useState(DEFAULT_ENCYCLOPEDIA);
  const [timeline, setTimeline] = useState(DEFAULT_TIMELINE);
  const [diary, setDiary] = useState(DEFAULT_DIARY);

  // Search & Filter States
  const [searchQuery, setSearchQuery] = useState('');
  const [encyCategory, setEncyCategory] = useState('all');
  const [diaryMoodFilter, setDiaryMoodFilter] = useState('all');

  // Modals / View States
  const [selectedEncyItem, setSelectedEncyItem] = useState(null);
  const [selectedDiaryItem, setSelectedDiaryItem] = useState(null);
  
  // Create / Edit Modals
  const [showProfileModal, setShowProfileModal] = useState(false);
  const [showEncyModal, setShowEncyModal] = useState(false);
  const [showTimelineModal, setShowTimelineModal] = useState(false);
  const [showDiaryModal, setShowDiaryModal] = useState(false);

  // Temporary Form States
  const [profileForm, setProfileForm] = useState(DEFAULT_PROFILE);
  const [encyForm, setEncyForm] = useState({ id: '', title: '', category: 'about', icon: '🎀', summary: '', content: '' });
  const [timelineForm, setTimelineForm] = useState({ id: '', year: '', title: '', description: '', emoji: '✨' });
  const [diaryForm, setDiaryForm] = useState({ id: '', title: '', content: '', date: new Date().toISOString().split('T')[0], mood: '💖' });

  // Custom Toast Message
  const [toastMessage, setToastMessage] = useState('');

  const showToast = (msg) => {
    setToastMessage(msg);
    setTimeout(() => setToastMessage(''), 3000);
  };

  useEffect(() => {
    if (!auth) {
      setLoading(false);
      return;
    }

    const initAuth = async () => {
      try {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }
      } catch (err) {
        console.error("Auth initialization error:", err);
      }
    };

    initAuth();
    const unsubscribe = onAuthStateChanged(auth, (currentUser) => {
      setUser(currentUser);
      if (!currentUser) setLoading(false);
    });

    return () => unsubscribe();
  }, []);

  useEffect(() => {
    if (!user || !db) return;

    // 1. Profile Doc Listener
    const profileRef = doc(db, 'artifacts', appId, 'users', user.uid, 'memora_data', 'profile');
    const unsubProfile = onSnapshot(profileRef, (snap) => {
      if (snap.exists()) setProfile(snap.data());
    }, (err) => console.error("Profile snap error:", err));

    // 2. Settings Doc Listener
    const settingsRef = doc(db, 'artifacts', appId, 'users', user.uid, 'memora_data', 'settings');
    const unsubSettings = onSnapshot(settingsRef, (snap) => {
      if (snap.exists()) setSettings(snap.data());
    }, (err) => console.error("Settings snap error:", err));

    // 3. Encyclopedia Collection Listener
    const encyCol = collection(db, 'artifacts', appId, 'users', user.uid, 'memora_encyclopedia');
    const unsubEncy = onSnapshot(encyCol, (snap) => {
      if (!snap.empty) {
        const items = snap.docs.map(d => ({ id: d.id, ...d.data() }));
        items.sort((a, b) => (b.createdAt || 0) - (a.createdAt || 0));
        setEncyclopedia(items);
      }
    }, (err) => console.error("Ency snap error:", err));

    // 4. Timeline Collection Listener
    const timeCol = collection(db, 'artifacts', appId, 'users', user.uid, 'memora_timeline');
    const unsubTime = onSnapshot(timeCol, (snap) => {
      if (!snap.empty) {
        const items = snap.docs.map(d => ({ id: d.id, ...d.data() }));
        items.sort((a, b) => (a.createdAt || 0) - (b.createdAt || 0));
        setTimeline(items);
      }
    }, (err) => console.error("Timeline snap error:", err));

    // 5. Diary Collection Listener
    const diaryCol = collection(db, 'artifacts', appId, 'users', user.uid, 'memora_diary');
    const unsubDiary = onSnapshot(diaryCol, (snap) => {
      if (!snap.empty) {
        const items = snap.docs.map(d => ({ id: d.id, ...d.data() }));
        items.sort((a, b) => (b.createdAt || 0) - (a.createdAt || 0));
        setDiary(items);
      }
      setLoading(false);
    }, (err) => {
      console.error("Diary snap error:", err);
      setLoading(false);
    });

    return () => {
      unsubProfile();
      unsubSettings();
      unsubEncy();
      unsubTime();
      unsubDiary();
    };
  }, [user]);

  const saveProfileToDb = async (newProfile) => {
    setProfile(newProfile);
    showToast("Profil berhasil diperbarui! ✨");
    if (user && db) {
      try {
        const ref = doc(db, 'artifacts', appId, 'users', user.uid, 'memora_data', 'profile');
        await setDoc(ref, newProfile, { merge: true });
      } catch (e) {
        console.error("Save profile error:", e);
      }
    }
  };

  const saveSettingsToDb = async (newSettings) => {
    setSettings(newSettings);
    showToast("Pengaturan tampilan disimpan! 🌸");
    if (user && db) {
      try {
        const ref = doc(db, 'artifacts', appId, 'users', user.uid, 'memora_data', 'settings');
        await setDoc(ref, newSettings, { merge: true });
      } catch (e) {
        console.error("Save settings error:", e);
      }
    }
  };

  const saveEncyEntry = async (e) => {
    e.preventDefault();
    if (!encyForm.title.trim()) return;

    const payload = {
      title: encyForm.title,
      category: encyForm.category,
      icon: encyForm.icon || '🎀',
      summary: encyForm.summary,
      content: encyForm.content,
      createdAt: Date.now()
    };

    if (encyForm.id) {
      setEncyclopedia(prev => prev.map(item => item.id === encyForm.id ? { ...item, ...payload } : item));
      if (user && db) {
        const ref = doc(db, 'artifacts', appId, 'users', user.uid, 'memora_encyclopedia', encyForm.id);
        await setDoc(ref, payload, { merge: true });
      }
      showToast("Artikel ensiklopedia diperbarui! 📖");
    } else {
      const newId = 'ency-' + Date.now();
      const newItem = { id: newId, ...payload };
      setEncyclopedia(prev => [newItem, ...prev]);
      if (user && db) {
        const col = collection(db, 'artifacts', appId, 'users', user.uid, 'memora_encyclopedia');
        await addDoc(col, payload);
      }
      showToast("Artikel baru ditambahkan! 🌸");
    }

    setShowEncyModal(false);
    setEncyForm({ id: '', title: '', category: 'about', icon: '🎀', summary: '', content: '' });
  };

  const deleteEncyEntry = async (id) => {
    setEncyclopedia(prev => prev.filter(i => i.id !== id));
    if (selectedEncyItem?.id === id) setSelectedEncyItem(null);
    showToast("Artikel dihapus 🗑️");
    if (user && db) {
      try {
        const ref = doc(doc(db, 'artifacts', appId, 'users', user.uid, 'memora_encyclopedia', id));
        await deleteDoc(ref);
      } catch (e) { console.error("Delete ency error:", e); }
    }
  };

  const saveTimelineEntry = async (e) => {
    e.preventDefault();
    if (!timelineForm.title.trim()) return;

    const payload = {
      year: timelineForm.year || '2026',
      title: timelineForm.title,
      description: timelineForm.description,
      emoji: timelineForm.emoji || '✨',
      createdAt: Date.now()
    };

    if (timelineForm.id) {
      setTimeline(prev => prev.map(i => i.id === timelineForm.id ? { ...i, ...payload } : i));
      if (user && db) {
        const ref = doc(db, 'artifacts', appId, 'users', user.uid, 'memora_timeline', timelineForm.id);
        await setDoc(ref, payload, { merge: true });
      }
      showToast("Timeline diperbarui! ⏳");
    } else {
      const newId = 'time-' + Date.now();
      setTimeline(prev => [...prev, { id: newId, ...payload }]);
      if (user && db) {
        const col = collection(db, 'artifacts', appId, 'users', user.uid, 'memora_timeline');
        await addDoc(col, payload);
      }
      showToast("Momen baru dicatat di timeline! 💖");
    }

    setShowTimelineModal(false);
    setTimelineForm({ id: '', year: '', title: '', description: '', emoji: '✨' });
  };

  const deleteTimelineEntry = async (id) => {
    setTimeline(prev => prev.filter(i => i.id !== id));
    showToast("Momen timeline dihapus 🗑️");
    if (user && db) {
      try {
        await deleteDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'memora_timeline', id));
      } catch (e) { console.error("Delete time error:", e); }
    }
  };

  const saveDiaryEntry = async (e) => {
    e.preventDefault();
    if (!diaryForm.title.trim()) return;

    const payload = {
      title: diaryForm.title,
      content: diaryForm.content,
      date: diaryForm.date || new Date().toISOString().split('T')[0],
      mood: diaryForm.mood || '💖',
      createdAt: Date.now()
    };

    if (diaryForm.id) {
      setDiary(prev => prev.map(i => i.id === diaryForm.id ? { ...i, ...payload } : i));
      if (user && db) {
        const ref = doc(db, 'artifacts', appId, 'users', user.uid, 'memora_diary', diaryForm.id);
        await setDoc(ref, payload, { merge: true });
      }
      showToast("Catatan harian diperbarui! 🎀");
    } else {
      const newId = 'diary-' + Date.now();
      setDiary(prev => [{ id: newId, ...payload }, ...prev]);
      if (user && db) {
        const col = collection(db, 'artifacts', appId, 'users', user.uid, 'memora_diary');
        await addDoc(col, payload);
      }
      showToast("Catatan harian berhasil disimpan! ✨");
    }

    setShowDiaryModal(false);
    setDiaryForm({ id: '', title: '', content: '', date: new Date().toISOString().split('T')[0], mood: '💖' });
  };

  const deleteDiaryEntry = async (id) => {
    setDiary(prev => prev.filter(i => i.id !== id));
    if (selectedDiaryItem?.id === id) setSelectedDiaryItem(null);
    showToast("Catatan harian dihapus 🗑️");
    if (user && db) {
      try {
        await deleteDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'memora_diary', id));
      } catch (e) { console.error("Delete diary error:", e); }
    }
  };

  const activeTheme = THEMES[settings.themePreset] || THEMES.pink;

  // Filtered Encyclopedia
  const filteredEncyclopedia = encyclopedia.filter(item => {
    const matchesCat = encyCategory === 'all' || item.category === encyCategory;
    const matchesSearch = item.title.toLowerCase().includes(searchQuery.toLowerCase()) || 
                          item.summary.toLowerCase().includes(searchQuery.toLowerCase());
    return matchesCat && matchesSearch;
  });

  // Filtered Diary
  const filteredDiary = diary.filter(item => {
    const matchesMood = diaryMoodFilter === 'all' || item.mood === diaryMoodFilter;
    const matchesSearch = item.title.toLowerCase().includes(searchQuery.toLowerCase()) || 
                          item.content.toLowerCase().includes(searchQuery.toLowerCase());
    return matchesMood && matchesSearch;
  });

  return (
    <div 
      className="min-h-screen w-full font-serif transition-colors duration-300 selection:bg-[#e98fb0] selection:text-white"
      style={{ backgroundColor: activeTheme.bg, color: activeTheme.textDark }}
    >
      {/* Top Cute Accent Bar */}
      <div 
        className="h-2 w-full transition-all duration-300" 
        style={{ background: `linear-gradient(90deg, ${activeTheme.accent}, ${activeTheme.accentDark}, #f8bbd0)` }}
      />

      {/* Floating Toast Alert */}
      {toastMessage && (
        <div className="fixed bottom-6 right-6 z-50 animate-bounce bg-white border-2 border-[#e98fb0] px-5 py-3 rounded-full shadow-lg flex items-center gap-2 text-sm font-sans font-bold text-[#c95d87]">
          <Sparkles className="w-4 h-4 text-[#e98fb0]" />
          {toastMessage}
        </div>
      )}

      {}
      <header className="sticky top-0 z-40 bg-white/90 backdrop-blur-md border-b" style={{ borderColor: activeTheme.line }}>
        <div className="max-w-6xl mx-auto px-4 py-3 flex flex-col md:flex-row items-center justify-between gap-4">
          {/* Brand Logo & Name */}
          <div className="flex items-center gap-3 cursor-pointer" onClick={() => setActiveTab('beranda')}>
            <div 
              className="w-11 h-11 rounded-full flex items-center justify-center font-bold text-xl shadow-sm border-2 animate-pulse"
              style={{ borderColor: activeTheme.accentDark, color: activeTheme.accentDark, backgroundColor: activeTheme.softBg }}
            >
              M
            </div>
            <div>
              <div className="flex items-center gap-2">
                <span className="font-bold text-2xl tracking-widest font-serif" style={{ color: activeTheme.textDark }}>
                  MEMORA
                </span>
                <span className="text-xs px-2 py-0.5 rounded-full font-sans font-bold text-white bg-[#e98fb0]">
                  v2.0
                </span>
              </div>
              <p className="text-xs font-sans tracking-wide text-gray-500 capitalize">
                {settings.tagline}
              </p>
            </div>
          </div>

          {/* Navigation Links */}
          <nav className="flex items-center gap-1 sm:gap-2 font-sans text-xs sm:text-sm font-semibold">
            <button
              onClick={() => { setActiveTab('beranda'); setSelectedEncyItem(null); setSelectedDiaryItem(null); }}
              className={`px-3 py-2 rounded-full flex items-center gap-1.5 transition-all ${
                activeTab === 'beranda' ? 'text-white shadow-md scale-105' : 'text-gray-600 hover:bg-pink-50'
              }`}
              style={{ backgroundColor: activeTab === 'beranda' ? activeTheme.accentDark : 'transparent' }}
            >
              <Home className="w-4 h-4" /> Beranda
            </button>

            <button
              onClick={() => { setActiveTab('ensiklopedia'); setSelectedEncyItem(null); }}
              className={`px-3 py-2 rounded-full flex items-center gap-1.5 transition-all ${
                activeTab === 'ensiklopedia' ? 'text-white shadow-md scale-105' : 'text-gray-600 hover:bg-pink-50'
              }`}
              style={{ backgroundColor: activeTab === 'ensiklopedia' ? activeTheme.accentDark : 'transparent' }}
            >
              <BookOpen className="w-4 h-4" /> Ensiklopedia
            </button>

            <button
              onClick={() => { setActiveTab('buku_harian'); setSelectedDiaryItem(null); }}
              className={`px-3 py-2 rounded-full flex items-center gap-1.5 transition-all ${
                activeTab === 'buku_harian' ? 'text-white shadow-md scale-105' : 'text-gray-600 hover:bg-pink-50'
              }`}
              style={{ backgroundColor: activeTab === 'buku_harian' ? activeTheme.accentDark : 'transparent' }}
            >
              <BookHeart className="w-4 h-4" /> Buku Harian
            </button>

            <button
              onClick={() => setActiveTab('pengaturan')}
              className={`px-3 py-2 rounded-full flex items-center gap-1.5 transition-all ${
                activeTab === 'pengaturan' ? 'text-white shadow-md scale-105' : 'text-gray-600 hover:bg-pink-50'
              }`}
              style={{ backgroundColor: activeTab === 'pengaturan' ? activeTheme.accentDark : 'transparent' }}
            >
              <Settings className="w-4 h-4" /> Customizer
            </button>
          </nav>
        </div>
      </header>

      {/* Main Content Area */}
      <main className="max-w-6xl mx-auto px-4 py-8">

        {/* Welcome Custom Banner */}
        {settings.bannerText && (
          <div 
            className="mb-8 p-4 rounded-2xl border flex items-center justify-between gap-4 font-sans text-xs sm:text-sm shadow-sm animate-fade-in"
            style={{ backgroundColor: activeTheme.softBg, borderColor: activeTheme.line }}
          >
            <div className="flex items-center gap-3">
              <span className="text-xl">🌸</span>
              <p className="font-medium" style={{ color: activeTheme.accentDark }}>{settings.bannerText}</p>
            </div>
            <button 
              onClick={() => setSettings(s => ({ ...s, bannerText: '' }))}
              className="text-gray-400 hover:text-gray-600 p-1"
            >
              <X className="w-4 h-4" />
            </button>
          </div>
        )}

        {}
        {activeTab === 'beranda' && (
          <div className="space-y-8 animate-fade-in">
            {/* Hero & Quick Facts Grid */}
            <div className="grid grid-cols-1 lg:grid-cols-3 gap-6 items-stretch">

                <div>
                  <label className="block text-gray-600 font-bold mb-1">Teks Banner Pengumuman</label>
                  <input 
                    type="text" 
                    value={settings.bannerText}
                    onChange={(e) => setSettings({ ...settings, bannerText: e.target.value })}
                    className="w-full p-2.5 rounded-xl border focus:outline-none focus:ring-2 focus:ring-pink-400"
                  />
                </div>

                <button 
                  onClick={() => saveSettingsToDb(settings)}
                  className="px-6 py-2.5 rounded-full bg-pink-500 text-white font-bold text-xs shadow-md hover:bg-pink-600 transition-all"
                >
                  Simpan Perubahan Teks
                </button>
              </div>
            </div>
          </div>
        )}

      </main>

      {}

      {/* Modal 1: Edit Profile Quick Facts */}
      {showProfileModal && (
        <div className="fixed inset-0 z-50 bg-black/40 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-white max-w-md w-full rounded-3xl p-6 shadow-2xl border space-y-4 font-sans animate-scale-up" style={{ borderColor: activeTheme.line }}>
            <div className="flex justify-between items-center border-b pb-2">
              <h3 className="font-bold text-lg flex items-center gap-2">
                <User className="w-5 h-5 text-pink-500" /> Edit Quick Facts Profil
              </h3>
              <button onClick={() => setShowProfileModal(false)}><X className="w-5 h-5 text-gray-400" /></button>
            </div>

            <div className="space-y-3 text-xs">
              <div>
                <label className="block font-bold mb-1">Nama Lengkap</label>
                <input type="text" value={profileForm.name} onChange={e => setProfileForm({ ...profileForm, name: e.target.value })} className="w-full p-2 border rounded-xl" />
              </div>
              <div>
                <label className="block font-bold mb-1">Nama Panggilan</label>
                <input type="text" value={profileForm.aka} onChange={e => setProfileForm({ ...profileForm, aka: e.target.value })} className="w-full p-2 border rounded-xl" />
              </div>
              <div>
                <label className="block font-bold mb-1">Status / Bio Singkat</label>
                <input type="text" value={profileForm.status} onChange={e => setProfileForm({ ...profileForm, status: e.target.value })} className="w-full p-2 border rounded-xl" />
              </div>
              <div>
                <label className="block font-bold mb-1">Tanggal Lahir</label>
                <input type="text" value={profileForm.born} onChange={e => setProfileForm({ ...profileForm, born: e.target.value })} className="w-full p-2 border rounded-xl" />
              </div>
              <div>
                <label className="block font-bold mb-1">Warna Favorit</label>
                <input type="text" value={profileForm.color} onChange={e => setProfileForm({ ...profileForm, color: e.target.value })} className="w-full p-2 border rounded-xl" />
              </div>
              <div>
                <label className="block font-bold mb-1">Makanan Favorit</label>
                <input type="text" value={profileForm.food} onChange={e => setProfileForm({ ...profileForm, food: e.target.value })} className="w-full p-2 border rounded-xl" />
              </div>
              <div>
                <label className="block font-bold mb-1">Hobi</label>
                <input type="text" value={profileForm.hobby} onChange={e => setProfileForm({ ...profileForm, hobby: e.target.value })} className="w-full p-2 border rounded-xl" />
              </div>
            </div>

            <div className="flex gap-2 pt-2">
              <button onClick={() => { saveProfileToDb(profileForm); setShowProfileModal(false); }} className="flex-1 py-2 rounded-full bg-pink-500 text-white font-bold text-xs">Simpan Profil</button>
              <button onClick={() => setShowProfileModal(false)} className="px-4 py-2 rounded-full border text-xs">Batal</button>
            </div>
          </div>
        </div>
      )}

      {/* Modal 2: Add/Edit Encyclopedia Article */}
      {showEncyModal && (
        <div className="fixed inset-0 z-50 bg-black/40 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-white max-w-lg w-full rounded-3xl p-6 shadow-2xl border space-y-4 font-sans animate-scale-up" style={{ borderColor: activeTheme.line }}>
            <div className="flex justify-between items-center border-b pb-2">
              <h3 className="font-bold text-lg flex items-center gap-2">
                <BookOpen className="w-5 h-5 text-pink-500" /> {encyForm.id ? 'Edit Artikel' : 'Artikel Ensiklopedia Baru'}
              </h3>
              <button onClick={() => setShowEncyModal(false)}><X className="w-5 h-5 text-gray-400" /></button>
            </div>

            <form onSubmit={saveEncyEntry} className="space-y-3 text-xs">
              <div>
                <label className="block font-bold mb-1">Judul Artikel</label>
                <input type="text" value={encyForm.title} onChange={e => setEncyForm({ ...encyForm, title: e.target.value })} required className="w-full p-2 border rounded-xl" placeholder="Contoh: Kebiasaan Unikku" />
              </div>

              <div className="grid grid-cols-2 gap-2">
                <div>
                  <label className="block font-bold mb-1">Kategori</label>
                  <select value={encyForm.category} onChange={e => setEncyForm({ ...encyForm, category: e.target.value })} className="w-full p-2 border rounded-xl bg-white">
                    <option value="about">Tentang Saya 🎀</option>
                    <option value="lore">Kisah & Lore 📖</option>
                    <option value="favorites">Favorit 🎮</option>
                  </select>
                </div>
                <div>
                  <label className="block font-bold mb-1">Ikon / Emoji</label>
                  <input type="text" value={encyForm.icon} onChange={e => setEncyForm({ ...encyForm, icon: e.target.value })} className="w-full p-2 border rounded-xl" placeholder="🎀 / 🌸 / 🎮" />
                </div>
              </div>

              <div>
                <label className="block font-bold mb-1">Ringkasan Singkat</label>
                <input type="text" value={encyForm.summary} onChange={e => setEncyForm({ ...encyForm, summary: e.target.value })} className="w-full p-2 border rounded-xl" placeholder="Ringkasan 1 kalimat..." />
              </div>

              <div>
                <label className="block font-bold mb-1">Isi Lengkap Artikel</label>
                <textarea value={encyForm.content} onChange={e => setEncyForm({ ...encyForm, content: e.target.value })} rows={5} className="w-full p-2 border rounded-xl" placeholder="Tulis artikel selengkapnya di sini..." />
              </div>

              <div className="flex gap-2 pt-2">
                <button type="submit" className="flex-1 py-2 rounded-full bg-pink-500 text-white font-bold text-xs">Simpan Artikel</button>
                <button type="button" onClick={() => setShowEncyModal(false)} className="px-4 py-2 rounded-full border text-xs">Batal</button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* Modal 3: Add/Edit Timeline Event */}
      {showTimelineModal && (
        <div className="fixed inset-0 z-50 bg-black/40 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-white max-w-md w-full rounded-3xl p-6 shadow-2xl border space-y-4 font-sans animate-scale-up" style={{ borderColor: activeTheme.line }}>
            <div className="flex justify-between items-center border-b pb-2">
              <h3 className="font-bold text-lg flex items-center gap-2">
                <Clock className="w-5 h-5 text-pink-500" /> {timelineForm.id ? 'Edit Momen Timeline' : 'Tambah Momen Timeline'}
              </h3>
              <button onClick={() => setShowTimelineModal(false)}><X className="w-5 h-5 text-gray-400" /></button>
            </div>

            <form onSubmit={saveTimelineEntry} className="space-y-3 text-xs">
              <div className="grid grid-cols-2 gap-2">
                <div>
                  <label className="block font-bold mb-1">Tahun / Waktu</label>
                  <input type="text" value={timelineForm.year} onChange={e => setTimelineForm({ ...timelineForm, year: e.target.value })} required className="w-full p-2 border rounded-xl" placeholder="Contoh: 2024" />
                </div>
                <div>
                  <label className="block font-bold mb-1">Emoji / Simbol</label>
                  <input type="text" value={timelineForm.emoji} onChange={e => setTimelineForm({ ...timelineForm, emoji: e.target.value })} className="w-full p-2 border rounded-xl" placeholder="🌱 / ✨ / ✈️" />
                </div>
              </div>

              <div>
                <label className="block font-bold mb-1">Judul Momen</label>
                <input type="text" value={timelineForm.title} onChange={e => setTimelineForm({ ...timelineForm, title: e.target.value })} required className="w-full p-2 border rounded-xl" placeholder="Judul petualangan..." />
              </div>

              <div>
                <label className="block font-bold mb-1">Deskripsi Singkat</label>
                <textarea value={timelineForm.description} onChange={e => setTimelineForm({ ...timelineForm, description: e.target.value })} rows={3} className="w-full p-2 border rounded-xl" placeholder="Cerita singkat..." />
              </div>

              <div className="flex gap-2 pt-2">
                <button type="submit" className="flex-1 py-2 rounded-full bg-pink-500 text-white font-bold text-xs">Simpan Momen</button>
                <button type="button" onClick={() => setShowTimelineModal(false)} className="px-4 py-2 rounded-full border text-xs">Batal</button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* Modal 4: Add/Edit Diary Entry */}
      {showDiaryModal && (
        <div className="fixed inset-0 z-50 bg-black/40 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-white max-w-lg w-full rounded-3xl p-6 shadow-2xl border space-y-4 font-sans animate-scale-up" style={{ borderColor: activeTheme.line }}>
            <div className="flex justify-between items-center border-b pb-2">
              <h3 className="font-bold text-lg flex items-center gap-2">
                <BookHeart className="w-5 h-5 text-pink-500" /> {diaryForm.id ? 'Edit Catatan Harian' : 'Tulis Catatan Harian'}
              </h3>
              <button onClick={() => setShowDiaryModal(false)}><X className="w-5 h-5 text-gray-400" /></button>
            </div>

            <form onSubmit={saveDiaryEntry} className="space-y-3 text-xs">
              <div>
                <label className="block font-bold mb-1">Judul Catatan</label>
                <input type="text" value={diaryForm.title} onChange={e => setDiaryForm({ ...diaryForm, title: e.target.value })} required className="w-full p-2 border rounded-xl font-serif text-sm" placeholder="Judul harimu..." />
              </div>

              <div className="grid grid-cols-2 gap-2">
                <div>
                  <label className="block font-bold mb-1">Tanggal</label>
                  <input type="date" value={diaryForm.date} onChange={e => setDiaryForm({ ...diaryForm, date: e.target.value })} className="w-full p-2 border rounded-xl bg-white" />
                </div>
                <div>
                  <label className="block font-bold mb-1">Mood / Suasana Hati</label>
                  <select value={diaryForm.mood} onChange={e => setDiaryForm({ ...diaryForm, mood: e.target.value })} className="w-full p-2 border rounded-xl bg-white">
                    <option value="🌸">🌸 Ceria / Damai</option>
                    <option value="💖">💖 Bahagia / Cinta</option>
                    <option value="✨">✨ Semangat / Bikin Kagum</option>
                    <option value="☕">☕ Santai / Coffee Time</option>
                    <option value="😴">😴 Lelah / Mengantuk</option>
                    <option value="🎮">🎮 Seru Main Game</option>
                    <option value="☁️">☁️ Melankolis / Galau</option>
                  </select>
                </div>
              </div>

              <div>
                <label className="block font-bold mb-1">Isi Curhatan / Catatan</label>
                <textarea value={diaryForm.content} onChange={e => setDiaryForm({ ...diaryForm, content: e.target.value })} rows={6} className="w-full p-2 border rounded-xl leading-relaxed" placeholder="Tuliskan harimu di sini secara bebas..." />
              </div>

              <div className="flex gap-2 pt-2">
                <button type="submit" className="flex-1 py-2 rounded-full bg-pink-500 text-white font-bold text-xs">Simpan ke Buku Harian</button>
                <button type="button" onClick={() => setShowDiaryModal(false)} className="px-4 py-2 rounded-full border text-xs">Batal</button>
              </div>
            </form>
          </div>
        </div>
      )}

      {}
      <footer className="mt-16 border-t bg-white py-8 text-center text-xs font-sans text-gray-400" style={{ borderColor: activeTheme.line }}>
        <div className="max-w-6xl mx-auto px-4 flex flex-col items-center gap-2">
          <div className="flex items-center gap-1 font-serif text-sm text-pink-600 font-bold">
            MEMORA ♡
          </div>
          <p>An encyclopedia of a sweet little life · Personal digital diary & archive</p>
          <p className="text-[10px] text-gray-300">Dibuat dengan kenangan manis 🌸</p>
        </div>
      </footer>
    </div>
  );
}
